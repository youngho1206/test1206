이 기종에서 게임 실행 시 실제로 적용되는 키보드 기본값입니다. 게임패드는 연결하면 바로 대응 버튼으로 작동합니다.
이 기본값을 다른 키로 바꾸고 싶다면, 화면 오른쪽 위의 ⚙ (설정) 아이콘 → Control Settings(조작 설정)에서 버튼을 클릭하고 원하는 키보드 키 또는 게임패드 버튼을 눌러 다시 지정할 수 있습니다.
const saveBtn=document.getElementById("saveState"); const loadBtn=document.getElementById("loadState"); saveBtn.onclick=()=>{if(!currentFile)return alert("게임 실행 후 저장하세요.");const name=currentFile.name.replace(/\.[^.]+$/,'')+"(게임기록저장).sav";if(window.EJS_emulator&&window.EJS_emulator.saveState)window.EJS_emulator.saveState();alert(name+" 이름으로 저장됩니다.")}; loadBtn.onclick=()=>{if(window.EJS_emulator&&window.EJS_emulator.loadState)window.EJS_emulator.loadState();else alert("저장 파일을 선택하세요.")}; document.getElementById("gameStart").onclick=()=>{if(currentFile)launch(currentFile);else alert("게임 파일 또는 폴더를 먼저 선택하세요.")};
